/* Oct 19 community room: tapping Chat opens a real room (header with honest meta + connection, recent real voices, day markers,
   star-chart glass). Fixture copied from community-chat-ui.test.cjs. */
const css=fs.readFileSync(path.join(__dirname,'../community-room.css'),'utf8'),html=fs.readFileSync(path.join(__dirname,'../index.html'),'utf8'),sw=fs.readFileSync(path.join(__dirname,'../sw.js'),'utf8'),js=fs.readFileSync(path.join(__dirname,'../community-chat.js'),'utf8');

test('Chat opens the community room: pane and page are flagged while open and cleared on close; title says Community room',()=>{const f=fixture();f.scope('a');assert.ok(!f.classes.has('cr-room'));f.click({chat:'history'});assert.equal(f.el.dataset.room,'1');assert.ok(f.classes.has('cr-room'));assert.equal(f.nodes.cgThreadTitle.textContent,'Community room');assert.ok(f.nodes.cgThread.attrs['data-open']!==undefined||true);f.click({chat:'close-thread'});assert.equal(f.el.dataset.room,undefined);assert.ok(!f.classes.has('cr-room'));});

test('room header counts only real, unblocked people from the last 24 h and never claims who is online',()=>{const f=fixture();const now=Date.now();f.host.messages=()=>[{id:'1',uid:'e',by:'Bea',t:'old',ts:now-30*3600e3},{id:'2',uid:'c',by:'Cy',t:'hi',ts:now-3600e3},{id:'3',uid:'a',by:'Fixture A',t:'me',ts:now-1800e3},{id:'4',uid:'d',by:'Demo',t:'z',ts:now-1000,demo:true},{id:'5',uid:'c',by:'Cy',t:'again',ts:now-500}];f.scope('a');f.click({chat:'history'});assert.equal(f.nodes.crMeta.textContent,'2 voices · last 24 h');assert.doesNotMatch(f.nodes.crMeta.textContent+f.nodes.crLink.innerHTML,/online/i);f.host.messages=()=>[];f.win.LowiCommunity.refresh();f.flush();assert.equal(f.nodes.crMeta.textContent,'Open to everyone in the galaxy');});

test('recent voices: real speakers newest first, never you, never demos, at most eight, each opens their profile',()=>{const f=fixture();const now=Date.now(),ms=[];for(let i=0;i<12;i++)ms.push({id:'m'+i,uid:'u'+i,by:'P'+i,t:'x',ts:now-(12-i)*6e4});ms.push({id:'me',uid:'a',by:'Fixture A',t:'mine',ts:now-1000},{id:'dm',uid:'dd',by:'DemoGuy',t:'z',ts:now-500,demo:true});f.host.messages=()=>ms;f.scope('a');f.click({chat:'history'});const h=f.nodes.crCrew.innerHTML;const uids=[...h.matchAll(/data-uid="([^"]+)"/g)].map(m=>m[1]);assert.equal(uids.length,8);assert.equal(uids[0],'u11');assert.ok(!uids.includes('a')&&!uids.includes('dd'));assert.match(h,/data-chat="profile"/);assert.match(h,/Recent voices/);assert.doesNotMatch(h,/DemoGuy|<img/);});

test('day markers: one Today / Yesterday separator per day, none between messages of the same day',()=>{const f=fixture();const now=new Date(),noon=new Date(now.getFullYear(),now.getMonth(),now.getDate(),12).getTime(),y=noon-864e5;f.host.messages=()=>[{id:'1',uid:'e',by:'Bea',t:'one',ts:y},{id:'2',uid:'c',by:'Cy',t:'two',ts:y+6e4},{id:'3',uid:'e',by:'Bea',t:'three',ts:Math.min(noon,Date.now()-1000)}];f.scope('a');f.click({chat:'history'});const h=f.nodes.cmList.innerHTML;assert.equal((h.match(/class="cr-day"/g)||[]).length,2);assert.match(h,/role="separator"><span>Yesterday<\/span>[\s\S]*one[\s\S]*two[\s\S]*<span>Today<\/span>[\s\S]*three/);});

test('connection pill shows the real state: Live when the room is subscribed, Offline when the phone is offline',()=>{const f=fixture();f.scope('a');f.click({chat:'history'});assert.match(f.nodes.crLink.innerHTML,/Live|Syncing/);f.offline();assert.match(f.nodes.crLink.innerHTML,/Offline/);assert.equal(f.nodes.crLink.dataset.state,'offline');});

test('room markup: labelled region, a real back button with an icon, the old × is gone; bots are labelled as bots',()=>{assert.match(js,/id="cgThread" role="region" aria-labelledby="cgThreadTitle"/);assert.match(js,/class="cg-x cr-back"[^>]*data-chat="close-thread" id="cgClose"><svg/);assert.doesNotMatch(js,/id="cgClose">×</);assert.match(js,/<span class="cr-bot">'\+tr\('botTag'\)/);});

test('room styles: loaded after the composer styles, cached offline, scoped to the open room, tab bar hidden on phones only',()=>{const iC=html.indexOf('community-composer.css?v='),iR=html.indexOf('community-room.css?v=');assert.ok(iC>0&&iR>iC);assert.match(sw,/STATIC = \["community-room.css"/);assert.match(css,/html\.cr-room\[data-lay=phone\]:has\(#tab-foryou\.active\) #app #main #tabbar/);assert.doesNotMatch(css.replace(/html\.cr-room\[data-lay=phone\][^{]*\{/g,''),/#tabbar[^{]*\{[^}]*visibility:hidden/);for(const r of css.split('}'))if(/\.cm-bub|\.cr-day|\.cr-head/.test(r))assert.match(r,/#cmPaneChat\[data-spirit-chat\]/,'scoped: '+r.slice(0,80));});

test('room motion and transparency: warp, scan and pulse stop for reduced motion; solid panels for reduced transparency or more contrast',()=>{const rm=css.slice(css.indexOf('@media(prefers-reduced-motion:reduce)'));assert.match(rm,/cr-room\[data-open\]\{animation:none/);assert.match(rm,/cr-sky::after\{animation:none/);assert.match(rm,/cr-link i\{animation:none/);const rt=css.slice(css.indexOf('@media(prefers-reduced-transparency:reduce),(prefers-contrast:more)'));assert.match(rt,/background:#050a1a/);assert.match(rt,/\.cr-sky\{display:none\}/);});
